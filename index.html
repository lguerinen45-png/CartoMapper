async function initClimat() { if(chart) chart.destroy(); const lat = parseFloat(document.getElementById('lat').value); const lon = parseFloat(document.getElementById('lon').value); // Données de secours pour Yaoundé (si l'API échoue) let precip = [10, 15, 40, 80, 120, 100, 80, 70, 90, 110, 40, 15]; let temp = [23, 24, 24, 23, 22, 21, 20, 20, 21, 22, 23, 23]; // Tentative de récupération des vraies données try { const url = `https://api.open-meteo.com/v1/forecast?latitude=${lat}&longitude=${lon}&daily=temperature_2m_max,precipitation_sum&past_days=365&timezone=auto`; const response = await fetch(url); const data = await response.json(); let precipMensuel = new Array(12).fill(0); let tempMensuelle = new Array(12).fill(0); let count = new Array(12).fill(0); data.daily.time.forEach((date, i) => { const mois = new Date(date).getMonth(); precipMensuel[mois] += data.daily.precipitation_sum[i] || 0; tempMensuelle[mois] += data.daily.temperature_2m_max[i] || 0; count[mois]++; }); precip = precipMensuel.map((p, i) => count[i] > 0 ? p : precip[i]); temp = tempMensuelle.map((t, i) => count[i] > 0 ? (t / count[i]).toFixed(1) : temp[i]); } catch (error) { console.log("Utilisation des données climatiques de secours"); } // Création du graphique avec un petit délai pour s'assurer que Chart.js est prêt setTimeout(() => { const ctx = document.getElementById('myChart').getContext('2d'); chart = new Chart(ctx, { type: 'bar', data: { labels: ['J','F','M','A','M','J','J','A','S','O','N','D'], datasets: [{ label: 'Précip. (mm)', data: precip, backgroundColor: '#3498db', yAxisID: 'y', order: 2 }, { label: 'Temp. max (°C)', data: temp, type: 'line', borderColor: '#e74c3c', backgroundColor: '#e74c3c', pointRadius: 3, yAxisID: 'y1', order: 1 }] }, options: { responsive: true, maintainAspectRatio: false, plugins: { legend: {position: 'bottom', labels: {boxWidth: 10, font: {size: 10}}} }, scales: { y: { position: 'left', title: {display: true, text: 'Précipitations (mm)', font: {size: 10}}, ticks: {font: {size: 9}} }, y1: { position: 'right', grid: {drawOnChartArea: false}, title: {display: true, text: 'Température (°C)', font: {size: 10}}, ticks: {font: {size: 9}} } } } }); }, 300); }
